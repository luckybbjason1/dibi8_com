Custom Annotators#

Supervision allows you to create custom annotators for specialized use cases. Here’s an example of a custom annotator that draws arrows: `````python import supervision as sv

class ArrowAnnotator(sv.Annotator): “““Custom annotator for drawing directional arrows.”””

def __init__(self, color=None, thickness=2): super().__init__()
    self.color = color or sv.Color.WHITE
    self.thickness = thickness

def annotate(self, scene, detections, direction="right"): for i, detection in enumerate(detections): center_x = int((detection.xyxy[0] + detection.xyxy[2]) / 2)
        center_y = int((detection.xyxy[1] + detection.xyxy[3]) / 2)
        
        if direction == "right": end_x = center_x + 50
            end_y = center_y
        elif direction == "left": end_x = center_x - 50
            end_y = center_y
        elif direction == "up": end_x = center_x
            end_y = center_y - 50
        else: end_x = center_x
            end_y = center_y + 50
        
        cv2.arrowedLine(
            scene,
            (center_x, center_y),
            (end_x, end_y),
            (self.color.r, self.color.g, self.color.b),
            self.thickness
        )
    
    return scene
